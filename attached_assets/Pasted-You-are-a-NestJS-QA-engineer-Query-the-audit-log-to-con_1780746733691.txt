real customer payment or refund.

"The 35 historical failed jobs that were drained this week were all
false-failures by inspection, but a quick confirmation that no real

════════════════════════════════════════════════════════

Run:
1. node -e "
const { PrismaClient } = require('@prisma/client')
const p = new PrismaClient()
// Find any payment or refund records created in the window
// when the 35 jobs were drained (this week, approximately)
const since = new Date(Date.now() - 7 * 24 * 60 * 60 * 1000)  // last 7 days
Promise.all([
  p.payment.count({ where: {
    type:       { in: ['refund','partial_refund'] },
    createdAt:  { gte: since },
  }}),
  p.order.count({ where: {
    status:    { in: ['cancelled'] },
    updatedAt: { gte: since },
  }}),
  p.auditLog.count({ where: {
    action:    { contains: 'refund' },
    createdAt: { gte: since },
  }}),
]).then(([refunds, cancelled, auditEntries]) => {
  console.log('Refunds issued this week:',       refunds)
  console.log('Orders cancelled this week:',     cancelled)
  console.log('Refund audit entries this week:',  auditEntries)
  p.\$disconnect()
})
"

2. node -e "
const { PrismaClient } = require('@prisma/client')
const p = new PrismaClient()
const since = new Date(Date.now() - 7 * 24 * 60 * 60 * 1000)
// Find any notification failures in the audit log
p.auditLog.findMany({
  where: {
    action:    { in: ['notification_failed', 'job_failed', 'refund_failed'] },
    createdAt: { gte: since },
  },
  orderBy: { createdAt: 'desc' },
  take: 50,
}).then(entries => {
  console.log('Failure audit entries this week:', entries.length)
  if (entries.length > 0) {
    console.log('First 10 entries:')
    entries.slice(0, 10).forEach(e => console.log(
      ' -', e.action, '|', e.entityType, e.entityId, '|', e.createdAt.toISOString()
    ))
  }
  p.\$disconnect()
})
"

3. node -e "
const { PrismaClient } = require('@prisma/client')
