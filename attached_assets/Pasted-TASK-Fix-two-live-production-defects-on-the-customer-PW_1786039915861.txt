model CateringEnquiry {
  id            String   @id @default(cuid())
  occasionType  String
  eventDate     DateTime
  guestCount    Int
  postcode      String
  budgetBand    String
  cuisines      String[]
  dietaryNotes  String?
  contactName   String

  contactPhone  String
  hearAboutUs   String?

  createdAt     DateTime @default(now())
  updatedAt     DateTime @updatedAt
