STEP 4 — Email alert when new chargeback arrives
File: apps/api/src/modules/notifications/templates/
      chargeback-alert.template.ts (create)

import { baseLayout, h2, p, brandButton } from './base-layout'

interface ChargebackAlertData {
  orderNumber:       string
  vendorName:        string
  customerEmail:     string
  amountPence:       number
  reason:            string
  evidenceDueBy:     string  // formatted date string
  daysRemaining:     number
  stripeDisputeUrl:  string
  adminChargebackUrl: string
}

export function chargebackAlertTemplate(
  data: ChargebackAlertData
): { subject: string; html: string } {
  const amount = `£${(data.amountPence / 100).toFixed(2)}`
  const urgency = data.daysRemaining <= 2 ? '🚨 URGENT: ' : '⚠️ '

  return {
    subject: `${urgency}New chargeback — Order #${data.orderNumber} — ${amount} — ${data.daysRemaining} days to respond`,
    html: baseLayout(
      'Chargeback alert',
      `
      ${h2(`New chargeback received`)}
      ${p(`A customer has raised a chargeback on order
           <strong>#${data.orderNumber}</strong> from
           <strong>${data.vendorName}</strong>.`)}

      <table style="width:100%;border-collapse:collapse;margin:16px 0;font-size:14px">
        ${[
          ['Order',           `#${data.orderNumber}`],
          ['Vendor',          data.vendorName],
          ['Customer',        data.customerEmail],
          ['Amount disputed', amount],
          ['Reason',          data.reason],
          ['Evidence due',    `${data.evidenceDueBy} (${data.daysRemaining} days remaining)`],
        ].map(([label, value]) => `
          <tr>
            <td style="padding:8px 12px;background:#F4F4F4;font-weight:600;color:#1A1A1A;width:40%">${label}</td>
            <td style="padding:8px 12px;color:#666666">${value}</td>
          </tr>
        `).join('')}
      </table>

      ${data.daysRemaining <= 2
        ? p(`<strong style="color:#E8220A">⚠️ Evidence must be submitted within
             ${data.daysRemaining} days. Missing this deadline results in an
             automatic loss.</strong>`)
        : p(`You have ${data.daysRemaining} days to submit evidence.`)}

      ${brandButton('Review chargeback in admin', data.adminChargebackUrl)}
      ${p(`<a href="${data.stripeDisputeUrl}" style="color:#666666;font-size:13px">
           View in Stripe Dashboard →</a>`)}
      `,
      `${amount} chargeback — ${data.daysRemaining} days to respond`
    ),
  }
}

case 'charge.dispute.created': {
  const dispute = event.data.object as Stripe.Dispute
  // ... existing chargeback creation logic ...

  // Send alert email to admin
  const evidenceDueBy = new Date(dispute.evidence_details.due_by * 1000)

STEP 7 — Reconcile order finances when chargeback is lost
File: apps/api/src/stripe/stripe-webhook.handler.ts

In the handler for charge.dispute.lost:

case 'charge.dispute.lost': {
  // Mark the chargeback as lost
  await this.prisma.chargeback.update({
    where:  { stripeDisputeId: dispute.id },
    data: {
      status:     'lost',
      outcome:    'lost',
      resolvedAt: new Date(),
    },
  })

  // Record the financial impact on the order
  await this.prisma.order.update({
    where: { id: chargeback.orderId },
    data: {
      chargebackLost:       true,
      chargebackAmountPence: dispute.amount,
    },
  })

  // Write audit log entry
  await this.prisma.auditLog.create({
    data: {
      action:     'chargeback_lost',
      entityType: 'Order',
      entityId:   chargeback.orderId,
      metadata: {
        stripeDisputeId: dispute.id,
        amountPence:     dispute.amount,
        reason:          dispute.reason,
      },
    },
  })

  // Alert admin of the loss
  await this.notificationsQueue.add('send', {
    key:        'chargeback_lost_alert',
    adminEmail: process.env.CHARGEBACK_ALERT_EMAIL ?? 'soul@feastpot.co.uk',
    data: {
      orderNumber: order.orderNumber,
      amountPence: dispute.amount,
    },
  })
  break
}
